// Toast Notification Component
const ToastNotification = ({ message, type = 'info', onClose }) => {
  const [isVisible, setIsVisible] = useState(true);
  
  useEffect(() => {
    const timer = setTimeout(() => {
      setIsVisible(false);
      setTimeout(onClose, 300);
    }, 3000);
    return () => clearTimeout(timer);
  }, [onClose]);
  
  const getIcon = () => {
    switch (type) {
      case 'success': return <FaCheck className="toast-icon" />;
      case 'error': return <FaTimesCircle className="toast-icon" />;
      case 'warning': return <FaExclamationTriangle className="toast-icon" />;
      default: return <FaInfoCircle className="toast-icon" />;
    }
  };
  
  const getTypeClass = () => {
    switch (type) {
      case 'success': return 'toast-success';
      case 'error': return 'toast-error';
      case 'warning': return 'toast-warning';
      default: return 'toast-info';
    }
  };
  
  return (
    <div className={`toast-notification ${getTypeClass()} ${isVisible ? 'toast-show' : 'toast-hide'}`}>
      {getIcon()}
      <span className="toast-message">{message}</span>
      <button className="toast-close" onClick={() => {
        setIsVisible(false);
        setTimeout(onClose, 300);
      }}>
        <FaTimes />
      </button>
    </div>
  );
};

// Toast Container Component
const ToastContainer = () => {
  const [toasts, setToasts] = useState([]);
  
  const addToast = (message, type = 'info') => {
    const id = Date.now();
    setToasts(prev => [...prev, { id, message, type }]);
  };
  
  const removeToast = (id) => {
    setToasts(prev => prev.filter(toast => toast.id !== id));
  };
  
  // Expose addToast globally
  useEffect(() => {
    window.showToast = addToast;
    return () => {
      delete window.showToast;
    };
  }, []);
  
  return (
    <div className="toast-container">
      {toasts.map(toast => (
        <ToastNotification
          key={toast.id}
          message={toast.message}
          type={toast.type}
          onClose={() => removeToast(toast.id)}
        />
      ))}
    </div>
  );
};

  };
};

// Custom hook for voice recognition
const useVoiceRecognition = (callback) => {
  const recognitionRef = useRef(null);
  const [isListening, setIsListening] = useState(false);
  
  useEffect(() => {
    if (typeof window !== 'undefined') {
      const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
      if (SpeechRecognition) {
        recognitionRef.current = new SpeechRecognition();
        recognitionRef.current.continuous = false;
        recognitionRef.current.interimResults = false;
        recognitionRef.current.lang = 'en-US';
        
        recognitionRef.current.onresult = (event) => {
          const transcript = event.results[0][0].transcript;
          callback(transcript);
        };
        
        recognitionRef.current.onerror = (event) => {
          console.error('Speech recognition error', event.error);
          setIsListening(false);
          if (window.showToast) {
            window.showToast('Voice recognition error. Please try again.', 'error');
          }
        };
        
        recognitionRef.current.onend = () => {
          setIsListening(false);
        };
      }
    }
  }, [callback]);
  
  const startListening = () => {
    if (recognitionRef.current) {
      setIsListening(true);
      recognitionRef.current.start();
    } else {
      if (window.showToast) {
        window.showToast('Speech recognition is not supported in your browser.', 'error');
      }
    }
  };
  
  return { isListening, startListening };
};

// AI Content Generator Component
const AIContentGenerator = ({ toast }) => {
  const [topic, setTopic] = useState('');
  const [contentType, setContentType] = useState('quiz');
  const [generatedContent, setGeneratedContent] = useState('');
  const [isGenerating, setIsGenerating] = useState(false);
  
  const generateContent = async () => {
    if (!topic.trim()) {
      toast('Please enter a topic', 'warning');
      return;
    }
    
    setIsGenerating(true);
    
    try {
      const content = await AIService.generateContent(topic, contentType);
      setGeneratedContent(content);
      toast('Content generated successfully!', 'success');
    } catch (error) {
      toast(error.message, 'error');
    } finally {
      setIsGenerating(false);
    }
  };
  
  const downloadContent = () => {
    const element = document.createElement('a');
    const file = new Blob([generatedContent], { type: 'text/plain' });
    element.href = URL.createObjectURL(file);
    element.download = `${topic.replace(/\s+/g, '_')}_${contentType}.txt`;
    document.body.appendChild(element);
    element.click();
    document.body.removeChild(element);
    toast('Content downloaded!', 'success');
  };
  
  return (
    <div className="ai-content-generator">
      <div className="tool-header">
        <h3><FaRobot /> AI Content Generator</h3>
        <p>Generate educational content using AI</p>
      </div>
      
      <div className="tool-controls">
        <div className="control-group">
          <label>Topic:</label>
          <input
            type="text"
            value={topic}
            onChange={(e) => setTopic(e.target.value)}
            placeholder="Enter a topic..."
          />
        </div>
        
        <div className="control-group">
          <label>Content Type:</label>
          <select
            value={contentType}
            onChange={(e) => setContentType(e.target.value)}
          >
            <option value="quiz">Quiz</option>
            <option value="study">Study Material</option>
            <option value="questions">Important Questions</option>
            <option value="notes">Sticky Notes</option>
          </select>
        </div>
        
        <button
          onClick={generateContent}
          className="generate-btn"
          disabled={isGenerating}
        >
          {isGenerating ? 'Generating...' : 'Generate Content'}
        </button>
      </div>
      
      {generatedContent && (
        <div className="generated-content">
          <div className="content-header">
            <h4>Generated Content</h4>
            <button onClick={downloadContent} className="download-btn">
              <FaDownload /> Download
            </button>
          </div>
          <pre>{generatedContent}</pre>
        </div>
      )}
    </div>
  );
};

          const result = eval(code);
          mockOutput = String(result);
        } catch (e) {
          mockOutput = `Error: ${e.message}`;
        }
      } else if (language === 'python') {
        mockOutput = 'Python output would appear here\nThis is a simulation of Python code execution.';
      } else if (language === 'html') {
        mockOutput = 'HTML code would render in a browser\nThis is a simulation of HTML rendering.';
      } else if (language === 'css') {
        mockOutput = 'CSS styles would apply to HTML elements\nThis is a simulation of CSS styling.';
      } else if (language === 'c') {
        mockOutput = 'C code would be compiled and run\nThis is a simulation of C code execution.';
      }
      
      setOutput(mockOutput);
    } catch (error) {
      setOutput(`Error: ${error.message}`);
    } finally {
      setIsRunning(false);
    }
  };
  
  const getLanguageExtension = () => {
    switch (language) {
      case 'javascript': return javascript();
      case 'python': return python();
      case 'html': return html();
      case 'css': return css();
      case 'c': return () => {}; // Placeholder for C
      default: return javascript();
    }
  };
  
  const getTheme = () => {
    return theme === 'dark' ? oneDark : undefined;
  };
  
  return (
    <div className="code-editor-tab">
      <div className="section-header">
        <h2>Code Editor</h2>
      </div>
      
      <div className="code-editor-container">
        <div className="editor-controls">
          <div className="control-group">
            <label>Language:</label>
            <select
              value={language}
              onChange={(e) => setLanguage(e.target.value)}
            >
              {PROGRAMMING_LANGUAGES.map(lang => (
                <option key={lang.value} value={lang.value}>{lang.label}</option>
              ))}
            </select>
          </div>
          
          <div className="control-group">
            <label>Theme:</label>
            <select
              value={theme}
              onChange={(e) => setTheme(e.target.value)}
            >
              <option value="light">Light</option>
              <option value="dark">Dark</option>
            </select>
          </div>
          
          <button
            onClick={runCode}
            disabled={isRunning}
            className="run-code-btn"
          >
            {isRunning ? 'Running...' : 'Run Code'}
          </button>
        </div>
        
        <div className="editor-wrapper">

            value={code}
            height="400px"
            extensions={[getLanguageExtension()]}
            onChange={(value) => setCode(value)}

// Folder/File Explorer Component
const FileExplorer = ({ 
  folders, 
  files, 
  currentFolder, 
  setCurrentFolder, 
  createFolder, 
  createFile, 
  uploadFiles,
  deleteItem,
  downloadFile,
  previewFile
}) => {
  const [newFolderName, setNewFolderName] = useState('');
  const [newFileName, setNewFileName] = useState('');
  const [showCreateFolder, setShowCreateFolder] = useState(false);
  const [showCreateFile, setShowCreateFile] = useState(false);
  
  const handleCreateFolder = () => {
    if (newFolderName.trim()) {
      createFolder(newFolderName.trim(), currentFolder);
      setNewFolderName('');
      setShowCreateFolder(false);
    }
  };
  
  const handleCreateFile = () => {
    if (newFileName.trim()) {
      createFile(newFileName.trim(), currentFolder);
      setNewFileName('');
      setShowCreateFile(false);
    }
  };
  
  const handleUploadFiles = (e) => {
    uploadFiles(e.target.files, currentFolder);
  };
  
  const filteredFolders = currentFolder 
    ? folders.filter(folder => folder.parentId === currentFolder)
    : folders.filter(folder => !folder.parentId);
    
  const filteredFiles = currentFolder
    ? files.filter(file => file.folderId === currentFolder)
    : files.filter(file => !file.folderId);
  
  const currentFolderName = currentFolder 
    ? folders.find(f => f.id === currentFolder)?.name 
    : 'Root';

        <div className="breadcrumb">
          <button onClick={() => setCurrentFolder(null)}>Root</button>
          {currentFolder && (
            <>
              <span>/</span>
              <span>{currentFolderName}</span>
            </>
          )}
        </div>
        
        <div className="explorer-actions">
          <button onClick={() => setShowCreateFolder(true)} className="action-btn">
            <FaFolderPlus /> New Folder
          </button>
          <button onClick={() => setShowCreateFile(true)} className="action-btn">
            <FaFile /> New File
          </button>

                          ) : (

                              <div className="task-content">
                                <p className="task-title">{task.title}</p>
                                <p className="task-date">
                                  Due: {task.dueDate ? new Date(task.dueDate).toLocaleDateString() : 'No due date'}
                                </p>
                              </div>
                              <div className="task-item-actions">
                                <button
                                  onClick={() => startEditTask(status.key, index, task.title)}
                                  className="task-action-btn edit"
                                >
                                  <FaEdit />
                                </button>
                                <button
                                  onClick={() => deleteTask(status.key, index)}
                                  className="task-action-btn delete"
                                >
                                  <FaTrash />
                                </button>
                              </div>
                            </>
                          )}
                          
                          {!(editingTask?.status === status.key && editingTask?.index === index) && (
                            <div className="move-actions">
                              {TASK_STATUSES.filter(s => s.key !== status.key).map((targetStatus) => (
                                <button
                                  key={targetStatus.key}
                                  onClick={() => moveTask(status.key, targetStatus.key, index)}
                                  className="move-btn"
                                  style={{ '--target-color': targetStatus.color }}
                                >
                                  <FaArrowRight />
                                  <span>{targetStatus.label}</span>
                                </button>
                              ))}
                            </div>
                          )}
                        </div>
                      ))}
                      
                      {tasks[status.key].length === 0 && (
                        <div className="empty-column">
                          <p>No tasks yet</p>
                          <button
                            onClick={() => addTask(status.key)}
                            className="empty-add-btn"
                          >
                            <FaPlus />
                            Add first task
                          </button>
                        </div>
                      )}
                    </div>
                  </div>
                ))}
              </div>
            </div>
          )}
        </main>
        
        {/* Right Sidebar */}
        <RightSidebar 
          files={files} 
          resources={STUDY_RESOURCES.map(resource => ({
            ...resource,
            icon: resource.icon || FaBook,
            count: Math.floor(Math.random() * 20) + 1
          }))} 
          quizPerformanceData={quizPerformanceData}
          doughnutOptions={doughnutOptions}
        />
      </div>
      
      {/* Quiz Marks Modal */}
      {showQuizMarksModal && (
        <div className="modal-overlay">
          <div className="modal-content">
            <div className="modal-header">
              <h3>Select Quiz Marks</h3>
              <button onClick={() => setShowQuizMarksModal(false)} className="close-btn">
                <FaTimes />
              </button>
            </div>
            <div className="modal-body">
              <div className="marks-options">
                {[1, 2, 3, 5, 7, 10].map(mark => (
                  <button
                    key={mark}
                    onClick={() => {
                      setQuizMarks(mark);
                      generateQuizQuestions(selectedTopic, mark);
                    }}
                    className="mark-option-btn"
                  >
                    {mark} Mark{mark !== 1 ? 's' : ''}
                  </button>
                ))}
              </div>
            </div>
          </div>
        </div>
      )}
      
      {/* Study Material Modal */}
      {showStudyMaterialModal && (
        <div className="modal-overlay">
          <div className="modal-content">
            <div className="modal-header">
              <h3>Select Study Material Type</h3>
              <button onClick={() => setShowStudyMaterialModal(false)} className="close-btn">
                <FaTimes />
              </button>
            </div>
            <div className="modal-body">
              <div className="type-options">
                <button
                  onClick={() => {
                    setStudyMaterialType('simple');
                    generateStudyMaterial(selectedTopic, 'simple');
                  }}
                  className="type-option-btn"
                >
                  Simple Material
                </button>
                <button
                  onClick={() => {
                    setStudyMaterialType('detailed');
                    generateStudyMaterial(selectedTopic, 'detailed');
                  }}
                  className="type-option-btn"
                >
                  Detailed Material
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
      
      {/* Generated Study Material */}
      {generatedStudyMaterial && (
        <div className="modal-overlay">
          <div className="modal-content large">
            <div className="modal-header">
              <h3>AI-Generated Study Material: {selectedTopic}</h3>
              <button onClick={() => setGeneratedStudyMaterial('')} className="close-btn">
                <FaTimes />
              </button>
            </div>
            <div className="modal-body">
              <StudyMaterialGenerator
                generatedStudyMaterial={generatedStudyMaterial}
                handleSaveStudyMaterial={handleSaveStudyMaterial}
                handleDownloadStudyMaterial={handleDownloadStudyMaterial}
                setGeneratedStudyMaterial={setGeneratedStudyMaterial}
              />
            </div>
          </div>
        </div>
      )}
      
      {/* Questions Modal */}
      {showQuestionsModal && (
        <div className="modal-overlay">
          <div className="modal-content">
            <div className="modal-header">
              <h3>AI-Generated Important Questions: {selectedTopic}</h3>
              <button onClick={() => setShowQuestionsModal(false)} className="close-btn">
                <FaTimes />
              </button>
            </div>
            <div className="modal-body">
              <div className="questions-list">
                {generatedQuestions.map(question => (
                  <div key={question.id} className="question-item">
                    <p>{question.question}</p>
                    <div className="question-meta">
                      <span className="question-subject">{question.subject}</span>
                      <span className={`difficulty-tag ${question.difficulty.toLowerCase()}`}>
                        {question.difficulty}
                      </span>
                    </div>
                  </div>
                ))}
              </div>
              <div className="modal-actions">
                <button
                  onClick={() => {
                    // Add these questions to the important questions list
                    setImportantQuestions(prev => [...prev, ...generatedQuestions]);
                    setShowQuestionsModal(false);
                    toast('AI-generated questions added to your list!', 'success');
                  }}
                  className="primary-btn"
                >
                  Save Questions
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
      
      {/* Topic Options Modal */}
      {showTopicOptions && (
        <div className="modal-overlay">
          <div className="modal-content">
            <div className="modal-header">
              <h3>AI Options for: {selectedTopic}</h3>
              <button onClick={() => setShowTopicOptions(false)} className="close-btn">
                <FaTimes />
              </button>
            </div>
            <div className="modal-body">
              <div className="topic-options">
                <button onClick={() => handleTopicOptionSelect('quiz')} className="topic-option-btn">
                  <FaBrain /> AI Quiz
                </button>
                <button onClick={() => handleTopicOptionSelect('study')} className="topic-option-btn">
                  <FaBookOpen /> AI Study Material
                </button>
                <button onClick={() => handleTopicOptionSelect('questions')} className="topic-option-btn">
                  <FaQuestionCircle /> AI Important Questions
                </button>
                <button onClick={() => handleTopicOptionSelect('notes')} className="topic-option-btn">
                  <FaStickyNote /> AI Sticky Notes
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
      
      {/* File Preview Modal */}
      {showFilePreview && filePreview && (
        <FilePreview 
          file={filePreview} 
          onClose={() => setShowFilePreview(false)} 
        />
      )}
      
      {/* Event Modal */}
      {showEventModal && (
        <div className="modal-overlay">
          <div className="modal-content">
            <div className="modal-header">
              <h3>Add New Event</h3>
              <button onClick={() => setShowEventModal(false)} className="close-btn">
                <FaTimes />
              </button>
            </div>
            <div className="modal-body">
              <div className="form-group">
                <label>Event Title</label>
                <input
                  type="text"
                  value={newEvent.title}
                  onChange={(e) => setNewEvent({...newEvent, title: e.target.value})}
                  placeholder="Enter event title"
                />
              </div>
              <div className="form-group">
                <label>Date</label>
                <input
                  type="date"
                  value={newEvent.date}
                  onChange={(e) => setNewEvent({...newEvent, date: e.target.value})}
                />
              </div>
              <div className="form-group">
                <label>Type</label>
                <select
                  value={newEvent.type}
                  onChange={(e) => setNewEvent({...newEvent, type: e.target.value})}
                >
                  <option value="event">Event</option>
                  <option value="exam">Exam</option>
                  <option value="meeting">Meeting</option>
                  <option value="deadline">Deadline</option>
                </select>
              </div>
              <div className="form-group">
                <label>Duration (optional)</label>
                <input
                  type="text"
                  value={newEvent.duration}
                  onChange={(e) => setNewEvent({...newEvent, duration: e.target.value})}
                  placeholder="e.g., 2 hours, 30 minutes"
                />
              </div>
              <div className="modal-actions">
                <button onClick={addEvent} className="primary-btn">
                  Add Event
                </button>
              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default Dashboard;
